% Custom commands
\newcommand{\resumeItem}[1]{
  \item\small{
    {#1 \vspace{-1pt}}
  }
}

% {job title or degree}{dates}{organisation}{place}
\newcommand{\resumeSubheading}[4]{
  \item
    \begin{tabular*}{\linewidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & {\footnotesize\color{meta}#2} \\[-3pt]
      % Spans both columns, so a long second line (a thesis title) cannot push the dates past the margin.
      \multicolumn{2}{@{}l@{}}{\makebox[\linewidth][l]{\textit{\small\color{meta}#3}\hfill{\footnotesize\color{meta}#4}}} \\
    \end{tabular*}\vspace{-6pt}
}
